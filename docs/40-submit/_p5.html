adomain/record.py — 《entity》
from dataclasses import dataclass
from datetime import datetime

@dataclass
class Record:
    """台帳の 1 項目を表すエンティティ。"""
    _id: int
    _name: str
    _updated: datetime

    def __post_init__(self) -> None:
        self._updated = datetime.now()

    @property
    def name(self) -> str:
        return self._name

    @name.setter
    def name(self, value: str) -> None:
        if not value.strip():
            raise ValueError("名称は必須です")
        self._name = value
bservice/record_service.py — 《control》
from domain.record import Record
from repository.base import RecordRepository

class RecordService:
    """登録・検索のユースケースを実行する。"""
    def __init__(self, repo: RecordRepository) -> None:
        self._repo = repo

    def register(self, name: str) -> Record:
        record = Record(_id=self._repo.next_id(), _name=name)
        self._repo.save(record)
        return record

    def search(self, keyword: str) -> list[Record]:
        records = self._repo.all()
        return [r for r in records
                if keyword in r.name]
